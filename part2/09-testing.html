Testing

The model's judgment is Part 3's job. Everything around it is ordinary code, and every idea in this part gets a test that runs in a second, offline.

A cheap adversary for your gates

The point: harness ideas are deterministic and testable without a model. TestModel calls every tool; the approval gate either holds or it doesn't.

Saturday 21 NovBefore the agent runs unattended over the weekend, the team wants proof that nothing writes to the ledger without a person, whatever the model does.

TestModel is a stand-in model that calls every tool it's offered, with arguments generated from their schemas, then answers. It doesn't judge anything; it pokes everything.

from pydantic_ai import DeferredToolRequests
from pydantic_ai.models.test import TestModel

from billing import CLOSE_STARTS, FakeBilling
from billing_world import Deps, reminders

deps = Deps(billing=FakeBilling(), received=CLOSE_STARTS)
with reminders.override(model=TestModel()):  # calls every tool it's offered, with generated arguments
    result = reminders.run_sync('INV-2041 is overdue.', deps=deps)

assert isinstance(result.output, DeferredToolRequests)
print('waiting for a person:', sorted(call.tool_name for call in result.output.approvals))
print('tool calls that ran: ', [p.tool_name for m in result.all_messages() for p in m.parts if p.part_kind == 'tool-return'])
print('ledger writes:       ', deps.billing.writes)

One test per idea

Every chapter's idea is a property of your code, so each gets a test. FunctionModel lets a test play the model: look at what the request offered, answer badly on purpose, check what comes back. Set ALLOW_MODEL_REQUESTS = False once, and any test that would reach a real model fails instead.

code/part2/test_billing_agent.py
import asyncio

import pytest

from pydantic_ai import DeferredToolRequests, models
from pydantic_ai.messages import ModelMessage, ModelResponse, TextPart, ToolCallPart, ToolReturnPart
from pydantic_ai.models.function import AgentInfo, FunctionModel
from pydantic_ai.models.test import TestModel

from billing import CLOSE_STARTS, CORRECTION, CORRECTION_RECEIVED, FakeBilling
from billing_world import Deps, collections, reminders

models.ALLOW_MODEL_REQUESTS = False  # a test can never reach a real model, even by accident


def test_the_approval_gate_holds():
    """TestModel calls every tool with generated arguments: a cheap adversary for your wiring."""
    deps = Deps(billing=FakeBilling(), received=CLOSE_STARTS)
    with reminders.override(model=TestModel()):
        result = reminders.run_sync('INV-2041 is overdue.', deps=deps)
    assert isinstance(result.output, DeferredToolRequests)
    assert {call.tool_name for call in result.output.approvals} == {'mark_paid', 'record_payment'}
    assert deps.billing.writes == []  # nothing was written without a person


def test_close_hides_mark_paid_and_says_so():
    seen: list[AgentInfo] = []

    def look(messages: list[ModelMessage], info: AgentInfo) -> ModelResponse:
        seen.append(info)
        return ModelResponse(parts=[TextPart('No reminder during close.')])

    with reminders.override(model=FunctionModel(look)):
        reminders.run_sync('INV-2041?', deps=Deps(billing=FakeBilling(), received=CLOSE_STARTS, closing=True))
    assert 'mark_paid' not in [t.name for t in seen[0].function_tools]
    assert 'close is running' in seen[0].instructions


def test_installments_that_add_up_wrong_go_back():
    def too_much_then_right(messages: list[ModelMessage], info: AgentInfo) -> ModelResponse:
        rest = '4120.00' if len(messages) == 1 else '2060.00'
        plan = [{'amount': '2060.00', 'due_date': '2026-11-20'}, {'amount': rest, 'due_date': '2026-11-30'}]
        return ModelResponse(parts=[ToolCallPart(info.output_tools[0].name, {'is_commitment': True, 'invoice_number': 'INV-2041', 'installments': plan})])

    from ch5_answer_form import Deps as FormDeps, agent as form_agent

    with form_agent.override(model=FunctionModel(too_much_then_right)):
        result = form_agent.run_sync(CORRECTION, deps=FormDeps(FakeBilling(), CORRECTION_RECEIVED))
    assert sum(i.amount for i in result.output.installments) == 4120
    assert result.usage.requests == 2  # the validator sent it back once


def test_a_forged_lookup_never_reaches_the_model():
    from ch8_interfaces import what_the_model_reads

    assert asyncio.run(what_the_model_reads('server')).endswith('4120.00, open')


def test_a_forgotten_override_fails_loudly():
    with pytest.raises(RuntimeError, match='ALLOW_MODEL_REQUESTS'):
        collections.run_sync('INV-2041?', deps=Deps(billing=FakeBilling(), received=CLOSE_STARTS))
